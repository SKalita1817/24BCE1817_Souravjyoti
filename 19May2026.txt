	WORKER_ID INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
	FIRST_NAME CHAR(25),
	LAST_NAME CHAR(25),
	SALARY INT(15),
	JOINING_DATE DATETIME,
	DEPARTMENT CHAR(25)
);

SELECT department, COUNT(worker_id) AS employee_count
FROM worker
GROUP BY department;

SELECT department, AVG(salary) AS avg_department_salary
FROM worker
GROUP BY department;

SELECT department, MAX(salary) AS top_salary
FROM worker
GROUP BY department;

SELECT SUM(salary) AS total_salary
FROM worker
WHERE first_name LIKE '_a%';
